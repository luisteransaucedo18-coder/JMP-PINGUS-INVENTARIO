# Base de datos Supabase

Este documento contiene el modelo de datos necesario para reemplazar los datos simulados de la aplicación por Supabase.

Las imágenes del sistema **no se guardan en la base de datos ni en Supabase Storage**. Deben permanecer en `public/` o en otra carpeta local del proyecto. Por ese motivo, ninguna tabla contiene columnas de imagen, URL o archivo.

## Orden de instalación

1. Crea un proyecto vacío en Supabase.
2. Abre **SQL Editor > New query**.
3. Ejecuta completo el bloque **Migración principal**.
4. Ejecuta el bloque **Datos iniciales**.
5. Crea las cuentas desde **Authentication > Users** con los metadatos indicados en la sección correspondiente.
6. Ejecuta **Datos operativos de demostración** solamente si deseas conservar los datos del prototipo.

> No insertes contraseñas directamente en `auth.users`. Supabase Auth debe crear y cifrar las credenciales.

create or replace function public.registrar_auditoria()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_anterior jsonb;
  v_nuevo jsonb;
  v_registro_id text;
begin
  v_anterior := case when tg_op in ('UPDATE', 'DELETE') then to_jsonb(old) end;
  v_nuevo := case when tg_op in ('INSERT', 'UPDATE') then to_jsonb(new) end;
  v_registro_id := coalesce(
    v_nuevo ->> 'id', v_nuevo ->> 'sku', v_nuevo ->> 'codigo',
    v_anterior ->> 'id', v_anterior ->> 'sku', v_anterior ->> 'codigo'
  );

  insert into public.auditoria (
    usuario_id, tabla, registro_id, accion, datos_anteriores, datos_nuevos
  ) values (
    auth.uid(), tg_table_name, v_registro_id, tg_op, v_anterior, v_nuevo
  );

  if tg_op = 'DELETE' then
    return old;
  end if;
  return new;
end;
$$;

create or replace function public.crear_perfil_nuevo_usuario()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_rol public.rol_usuario;
  v_sede text;
begin
  v_rol := case new.raw_user_meta_data ->> 'rol'
    when 'gerente' then 'gerente'::public.rol_usuario
    when 'coordinador' then 'coordinador'::public.rol_usuario
    else 'analista'::public.rol_usuario
  end;

  v_sede := case new.raw_user_meta_data ->> 'sede'
    when 'Chimbote' then 'Chimbote'
    when 'Trujillo' then 'Trujillo'
    else 'Chiclayo'
  end;

  insert into public.perfiles (id, nombre, email, rol, sede)
  values (
    new.id,
    coalesce(nullif(new.raw_user_meta_data ->> 'nombre', ''), split_part(new.email, '@', 1)),
    new.email,
    v_rol,
    v_sede
  );
  return new;
end;
$$;

create trigger auth_usuario_creado
after insert on auth.users
for each row execute function public.crear_perfil_nuevo_usuario();

create or replace function public.recalcular_estado_material(p_sku text)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_total numeric(14,3);
  v_minimo numeric(14,3);
  v_estado public.estado_material;
begin
  select stock_minimo into v_minimo from public.materiales where sku = p_sku;
  select coalesce(sum(stock), 0) into v_total
  from public.inventario_sedes where material_sku = p_sku;

  v_estado := case
    when v_total = 0 then 'AGOTADO'::public.estado_material
    when v_total <= ceil(v_minimo * 0.30) then 'CRÍTICO'::public.estado_material
    when v_total <= v_minimo then 'BAJO'::public.estado_material
    else 'OK'::public.estado_material
  end;

  update public.materiales set estado = v_estado where sku = p_sku;
end;
$$;

create or replace function public.actualizar_estado_por_stock()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_sku text;
begin
  if tg_op = 'DELETE' then
    v_sku := old.material_sku;
  else
    v_sku := new.material_sku;
  end if;
  perform public.recalcular_estado_material(v_sku);
  if tg_op = 'DELETE' then
    return old;
  end if;
  return new;
end;
$$;

create trigger inventario_recalcular_estado
after insert or update or delete on public.inventario_sedes
for each row execute function public.actualizar_estado_por_stock();

create or replace function public.confirmar_requerimiento(
  p_requerimiento_id uuid,
  p_observaciones text default null
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_req public.requerimientos%rowtype;
  v_item record;
  v_stock numeric(14,3);
begin
  if public.rol_actual() <> 'coordinador' then
    raise exception 'Solo un coordinador puede confirmar un requerimiento';
  end if;

  select * into v_req
  from public.requerimientos
  where id = p_requerimiento_id
  for update;

  if not found or v_req.estado <> 'ENVIADO' then
    raise exception 'El requerimiento no existe o no está enviado';
  end if;

  for v_item in
    select * from public.requerimiento_items where requerimiento_id = p_requerimiento_id
  loop
    select stock into v_stock
    from public.inventario_sedes
    where material_sku = v_item.material_sku and sede = v_req.sede
    for update;

    v_stock := coalesce(v_stock, 0);
    if v_stock < v_item.cantidad then
      raise exception 'Stock insuficiente para % en %', v_item.material_sku, v_req.sede;
    end if;

    update public.inventario_sedes
    set stock = stock - v_item.cantidad
    where material_sku = v_item.material_sku and sede = v_req.sede;

    insert into public.movimientos_inventario (
      material_sku, sede, tipo, cantidad, stock_anterior, stock_nuevo,
      referencia_tipo, referencia_id, motivo, creado_por
    ) values (
      v_item.material_sku, v_req.sede, 'SALIDA_REQUERIMIENTO', -v_item.cantidad,
      v_stock, v_stock - v_item.cantidad, 'requerimiento', p_requerimiento_id,
      p_observaciones, auth.uid()
    );
  end loop;

  update public.requerimientos
  set estado = 'CONFIRMADO', observaciones = p_observaciones,
      confirmado_por = auth.uid(), fecha_confirmacion = current_date
  where id = p_requerimiento_id;
end;
$$;

create or replace function public.confirmar_compra(p_orden_id uuid, p_nota text default null)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_orden public.ordenes_compra%rowtype;
  v_item record;
  v_stock_anterior numeric(14,3);
begin
  if public.rol_actual() <> 'coordinador' then
    raise exception 'Solo un coordinador puede confirmar una compra';
  end if;

  select * into v_orden from public.ordenes_compra where id = p_orden_id for update;
  if not found or v_orden.estado <> 'APROBADO' then
    raise exception 'La orden no existe o no está aprobada';
  end if;

  for v_item in select * from public.orden_compra_items where orden_compra_id = p_orden_id loop
    select coalesce(max(stock), 0) into v_stock_anterior
    from public.inventario_sedes
    where material_sku = v_item.material_sku and sede = v_orden.sede;

    insert into public.inventario_sedes (material_sku, sede, stock)
    values (v_item.material_sku, v_orden.sede, v_item.cantidad_solicitada)
    on conflict (material_sku, sede)
    do update set stock = public.inventario_sedes.stock + excluded.stock;

    update public.materiales
    set precio_unitario = coalesce(v_item.precio_unitario, precio_unitario)
    where sku = v_item.material_sku;

    insert into public.movimientos_inventario (
      material_sku, sede, tipo, cantidad, stock_anterior, stock_nuevo,
      referencia_tipo, referencia_id, motivo, creado_por
    ) values (
      v_item.material_sku, v_orden.sede, 'INGRESO_COMPRA', v_item.cantidad_solicitada,
      v_stock_anterior, v_stock_anterior + v_item.cantidad_solicitada,
      'orden_compra', p_orden_id, p_nota, auth.uid()
    );
  end loop;

  update public.ordenes_compra
  set estado = 'COMPRADO', fecha_compra = current_date,
      coordinador_id = auth.uid(), nota_compra = p_nota
  where id = p_orden_id;
end;
$$;

create view public.v_inventario with (security_invoker = true) as
select
  m.sku as id,
  m.nombre,
  m.descripcion,
  c.nombre as categoria,
  m.unidad,
  m.marca,
  m.stock_minimo as minimo,
  m.precio_unitario,
  m.estado,
  jsonb_build_object(
    'Chiclayo', coalesce(max(i.stock) filter (where i.sede = 'Chiclayo'), 0),
    'Chimbote', coalesce(max(i.stock) filter (where i.sede = 'Chimbote'), 0),
    'Trujillo', coalesce(max(i.stock) filter (where i.sede = 'Trujillo'), 0)
  ) as stock_sedes,
  coalesce(sum(i.stock), 0) as stock_total
from public.materiales m
join public.categorias_material c on c.id = m.categoria_id
left join public.inventario_sedes i on i.material_sku = m.sku
where m.activo
group by m.sku, c.nombre;

create view public.v_requerimientos with (security_invoker = true) as
select
  r.*,
  p.codigo as proyecto_codigo,
  p.nombre as proyecto,
  a.nombre as analista,
  co.nombre as confirmado_por_nombre,
  coalesce(
    jsonb_agg(
      jsonb_build_object(
        'skuId', ri.material_sku,
        'nombre', ri.material_nombre,
        'cantidad', ri.cantidad
      ) order by ri.id
    ) filter (where ri.id is not null),
    '[]'::jsonb
  ) as materiales
from public.requerimientos r
join public.proyectos p on p.id = r.proyecto_id
join public.perfiles a on a.id = r.analista_id
left join public.perfiles co on co.id = r.confirmado_por

  a.nombre as analista,
  co.nombre as coordinador,

  coalesce(
    jsonb_agg(
      jsonb_build_object(
        'skuId', oci.material_sku,
        'nombre', oci.material_nombre,
        'cantidadSolicitada', oci.cantidad_solicitada,
        'precioUnitario', oci.precio_unitario,
        'subtotal', oci.subtotal
      ) order by oci.id
    ) filter (where oci.id is not null),
    '[]'::jsonb
  ) as items
from public.ordenes_compra oc
join public.perfiles a on a.id = oc.analista_id
left join public.perfiles co on co.id = oc.coordinador_id
left join public.orden_compra_items oci on oci.orden_compra_id = oc.id

## Consultas que utilizará el frontend

```sql
-- Inventario completo con stock agrupado por sede
select * from public.v_inventario order by id;

-- Requerimientos con sus materiales
select * from public.v_requerimientos order by fecha desc;

-- Órdenes de compra con items y total estimado
select * from public.v_compras order by fecha desc;

-- Confirmar recepción de una compra y aumentar el stock en una transacción
select public.confirmar_compra(
  'UUID_DE_LA_ORDEN'::uuid,
  'Factura F001-1234, materiales recibidos conformes'
);

-- Confirmar un requerimiento y descontar el stock de su sede en una transacción
select public.confirmar_requerimiento(
  'UUID_DEL_REQUERIMIENTO'::uuid,
  'Materiales aprobados para entrega'
);
```

## Correspondencia con el modelo React

| Modelo actual | Tabla o vista Supabase |
|---|---|
| `Usuario` | `perfiles` + `auth.users` |
| `Material` | `v_inventario` |
| `stockSedes` | `inventario_sedes` |
| `Proyecto` | `proyectos` |
| `Requerimiento` | `v_requerimientos` |
| `ReqMaterial` | `requerimiento_items` |
| `RequerimientoCompra` | `v_compras` |
| `CompraItem` | `orden_compra_items` |
| `Entrega` | `entregas` |
| `EntregaItem` | `entrega_items` |
| notificaciones generadas | consultas del estado + `notificaciones_leidas` |
| configuración | `configuracion_sistema` |

Los nombres de columnas en la base usan `snake_case`. Al mapearlos en TypeScript se pueden transformar a los nombres actuales, por ejemplo `precio_unitario` a `precioUnitario` y `stock_sedes` a `stockSedes`.
